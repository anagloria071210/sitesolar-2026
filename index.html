const speech = window.speechSynthesis; const readButton = document.getElementById('readButton'); const pauseButton = document.getElementById('pauseButton'); const stopButton = document.getElementById('stopButton'); const speechStatus = document.getElementById('speechStatus'); let chunks = []; let currentChunk = 0; let isReading = false; let session = 0; function getPageText() { const areas = document.querySelectorAll( 'main, .hero, body > section, footer' ); const texts = []; areas.forEach(area => { const copy = area.cloneNode(true); copy.querySelectorAll( 'button, input, select, textarea, script, style, svg, .accessibility-tools' ).forEach(el => el.remove()); const text = copy.innerText || copy.textContent; if (text.trim()) texts.push(text.trim()); }); return [...new Set(texts)].join('. ').replace(/\s+/g, ' '); } function splitText(text, limit = 220) { const sentences = text.match(/[^.!?]+[.!?]+|[^.!?]+$/g) || []; const result = []; let current = ''; sentences.forEach(sentence => { sentence = sentence.trim(); if (!sentence) return; if ((current + ' ' + sentence).trim().length > limit) { if (current) result.push(current); current = sentence; } else { current = (current + ' ' + sentence).trim(); } }); if (current) result.push(current); return result; } function getVoice() { const voices = speech.getVoices(); return voices.find(v => v.lang.toLowerCase() === 'pt-br') || voices.find(v => v.lang.toLowerCase().startsWith('pt')) || null; } function speakNext(id) { if (!isReading || id !== session) return; if (currentChunk >= chunks.length) { isReading = false; speechStatus.textContent = 'Leitura concluída!'; return; } const utterance = new SpeechSynthesisUtterance( chunks[currentChunk] ); utterance.lang = 'pt-BR'; utterance.rate = 0.95; utterance.pitch = 1; const voice = getVoice(); if (voice) utterance.voice = voice; utterance.onend = () => { if (id !== session || !isReading) return; currentChunk++; speechStatus.textContent = `Lendo parte ${currentChunk + 1} de ${chunks.length}...`; speakNext(id); }; utterance.onerror = (event) => { if (id !== session) return; if (event.error === 'canceled' || event.error === 'interrupted') return; isReading = false; speechStatus.textContent = 'Erro na leitura. Tente novamente.'; }; speech.speak(utterance); } readButton.addEventListener('click', () => { if (!('speechSynthesis' in window)) { speechStatus.textContent = 'Navegador sem suporte à leitura.'; return; } speech.cancel(); session++; const id = session; currentChunk = 0; isReading = true; const text = getPageText(); if (!text) { isReading = false; speechStatus.textContent = 'Nenhum texto encontrado.'; return; } chunks = splitText(text); speechStatus.textContent = 'Iniciando leitura...'; speakNext(id); }); pauseButton.addEventListener('click', () => { if (speech.speaking && !speech.paused) { speech.pause(); speechStatus.textContent = 'Leitura pausada.'; } else if (speech.paused) { speech.resume(); speechStatus.textContent = 'Continuando leitura...'; } else { speechStatus.textContent = 'Inicie a leitura primeiro.'; } }); stopButton.addEventListener('click', () => { session++; isReading = false; currentChunk = 0; speech.cancel(); speechStatus.textContent = 'Leitura parada.'; });
