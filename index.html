/* ================================================== LEITURA EM VOZ ALTA — VERSÃO COMPLETA ================================================== */ const speech = window.speechSynthesis; const readButton = document.getElementById('readButton'); const pauseButton = document.getElementById('pauseButton'); const stopButton = document.getElementById('stopButton'); const speechStatus = document.getElementById('speechStatus'); let speechChunks = []; let currentChunk = 0; let isReading = false; let speechSession = 0; /* Reúne o conteúdo textual do site */ function getPageText() { const areas = document.querySelectorAll( '.hero, body > section, footer' ); let texts = []; areas.forEach(area => { const copy = area.cloneNode(true); copy.querySelectorAll( 'button, input, select, textarea, script, style, svg, .accessibility-tools' ).forEach(el => el.remove()); const text = copy.innerText || copy.textContent; if (text.trim()) { texts.push(text.trim()); } }); return texts.join('. ') .replace(/\s+/g, ' ') .trim(); } /* Divide o texto em partes menores */ function splitText(text, maxLength = 220) { const sentences = text.match(/[^.!?]+[.!?]+|[^.!?]+$/g) || []; const chunks = []; let current = ''; sentences.forEach(sentence => { sentence = sentence.trim(); if (!sentence) return; if (sentence.length > maxLength) { if (current) { chunks.push(current); current = ''; } const words = sentence.split(' '); let part = ''; words.forEach(word => { if ((part + ' ' + word).trim().length > maxLength) { if (part) chunks.push(part); part = word; } else { part = (part + ' ' + word).trim(); } }); if (part) chunks.push(part); } else if ((current + ' ' + sentence).trim().length > maxLength) { if (current) chunks.push(current); current = sentence; } else { current = (current + ' ' + sentence).trim(); } }); if (current) chunks.push(current); return chunks; } /* Procura uma voz em português */ function getPortugueseVoice() { const voices = speech.getVoices(); return voices.find(voice => voice.lang.toLowerCase() === 'pt-br' ) || voices.find(voice => voice.lang.toLowerCase().startsWith('pt') ) || null; } /* Lê a próxima parte */ function speakNext(session) { if (!isReading || session !== speechSession) return; if (currentChunk >= speechChunks.length) { isReading = false; speechStatus.textContent = 'Leitura concluída!'; return; } const utterance = new SpeechSynthesisUtterance( speechChunks[currentChunk] ); utterance.lang = 'pt-BR'; utterance.rate = 0.95; utterance.pitch = 1; const voice = getPortugueseVoice(); if (voice) utterance.voice = voice; utterance.onend = function() { if (session !== speechSession || !isReading) return; currentChunk++; speechStatus.textContent = 'Lendo parte ' + (currentChunk + 1) + ' de ' + speechChunks.length + '...'; speakNext(session); }; utterance.onerror = function(event) { if (session !== speechSession) return; if (event.error === 'canceled' || event.error === 'interrupted') return; isReading = false; speechStatus.textContent = 'Ocorreu um erro na leitura. Tente novamente.'; }; speech.speak(utterance); } /* INICIAR LEITURA */ readButton.addEventListener('click', function() { if (!('speechSynthesis' in window)) { speechStatus.textContent = 'Seu navegador não oferece suporte à leitura em voz alta.'; return; } speech.cancel(); speechSession++; const session = speechSession; currentChunk = 0; isReading = true; const text = getPageText(); if (!text) { isReading = false; speechStatus.textContent = 'Não foi encontrado conteúdo para leitura.'; return; } speechChunks = splitText(text); speechStatus.textContent = 'Preparando leitura em português...'; /* Aguarda as vozes do navegador */ if (speech.getVoices().length === 0) { speechStatus.textContent = 'Carregando voz disponível...'; let attempts = 0; function waitForVoice() { if (session !== speechSession || !isReading) return; if (speech.getVoices().length > 0 || attempts >= 10) { speechStatus.textContent = 'Iniciando leitura...'; speakNext(session); } else { attempts++; setTimeout(waitForVoice, 300); } } waitForVoice(); } else { speakNext(session); } }); /* PAUSAR OU CONTINUAR */ pauseButton.addEventListener('click', function() { if (!('speechSynthesis' in window)) return; if (speech.speaking && !speech.paused) { speech.pause(); speechStatus.textContent = 'Leitura pausada.'; } else if (speech.paused) { speech.resume(); speechStatus.textContent = 'Continuando leitura...'; } else if (isReading) { speechStatus.textContent = 'Preparando leitura...'; } else { speechStatus.textContent = 'Clique em Ler página em voz alta primeiro.'; } }); /* PARAR LEITURA */ stopButton.addEventListener('click', function() { speechSession++; isReading = false; currentChunk = 0; speech.cancel(); speechStatus.textContent = 'Leitura parada.'; });
